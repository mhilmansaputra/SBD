-- DML & TABEL KHUSUS
--DML(AFTER / FOR) & TABEL INSERTED
CREATE  OR ALTER TRIGGER trg_TUnit_AuditInsert
ON TUnit
AFTER INSERT 
AS 
BEGIN
	SET NOCOUNT ON;--MENYEMBUNYIKAN PESAN
	SELECT
		kodeWil AS KodeBaru,
		Deskripsi AS DeskripsiBaru,
		Koordinator AS KoordinatorBaru
	FROM INSERTED;
END;
--uji coba
INSERT INTO TUnit values('A001', 'Buku Gramedia', 'Ipit')

--DML(AFTER / FOR) & TABEL DELETED
CREATE  OR ALTER TRIGGER trg_TUnit_AudiDelete
ON TUnit
AFTER DELETE 
AS 
BEGIN
	SET NOCOUNT ON;--MENYEMBUNYIKAN PESAN
	SELECT
		kodeWil AS KodeTerhapus,
		Deskripsi AS DeskripsiTerhapus
	FROM DELETED;
END;
--uji coba
DELETE FROM TUnit WHERE KodeWil='A001'

--DML(AFTER / FOR) & (UPDATE) TABEL INSERTED & DELETED
CREATE OR ALTER TRIGGER trg_TUnit_AuditUpdate
On TUnit
AFTER UPDATE
AS
BEGIN
	SET NOCOUNT ON;
	--membandingkan data baru dan data lama
	SELECT 
	d.KodeWil,
	d.Deskripsi AS DeskripsiLama,
	i.Deskripsi AS DeskripsiBaru,
	d.Koordinator As KoordinatorLama,
	i.Koordinator As KoordinatorBaru
	FROM DELETED d
	INNER JOIN INSERTED i ON d.KodeWil = i.Kodewil;
END;
--UJI COBA
UPDATE TUnit SET Deskripsi='Toko Buku Baru' WHERE KodeWil='A003'
UPDATE TUnit SET Koordinator='Suryo' WHERE KodeWil='A003'


--DML TRIGGER (INSTEAD OF)
--MEMBUAT VIEW YANG MENGGABUNGKAN INFORMASI BUKU (TMasterBuku)
--dan stok (TDetailMasterBuku)
CREATE OR ALTER VIEW vw_StockBukuTrigger
AS
SELECT
	m.IDBuku,
	m.JudulBuku,
	d.UNIT,
	d.instock
FROM TMasterbuku AS m
INNER JOIN TDetailMasterBukuUnit As d
On m.IDBuku = d. IDBuku;


--MENJALANKAN TRIGGER
SELECT * FROM vw_StockBukuTrigger;


--MEMBEUAT TRIGGER
CREATE OR ALTER TRIGGER trg_CegahHapusStock
ON vw_StockBukuTrigger
INSTEAD OF DELETE
AS
BEGIN
	SET NOCOUNT ON
	PRINT 'Penghapusasn Data Tidak Di Izinkan';
END;

--COBA HAPUS MELALUI VIEW 
DELETE FROM vw_StockBukuTrigger WHERE IDBuku = 1;
SELECT * FROM vw_StockBukuTrigger WHERE IDBuku = 1;
